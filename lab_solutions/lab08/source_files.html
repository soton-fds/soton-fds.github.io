library(shiny)
library(tidyverse)

ui <- fluidPage(
  titlePanel(
    tagList(
      textOutput("title", inline = TRUE),
      "Shiny App"
    ),
    # Without this, the browser tab would show the HTML of the title above.
    windowTitle = "Shiny App"
  ),
  textInput("name", label = "Enter your Name"),
  numericInput("age", label = "Enter your Age", value = 21),
  selectInput(
    "colour",
    label = "Choose your favourite colour.",
    choices = c(
      "Green" = "green",
      "Red" = "red",
      "Orange" = "orange"
    )
  ),
  textOutput("greeting"),
  uiOutput("coloured_greeting")
)

server <- function(input, output, session) {
  # req() leaves the output blank until a name has been entered, so the
  # heading reads just "Shiny App".
  output$title <- renderText({
    req(input$name)
    paste0(input$name, "'s")
  })

  # Both greetings use the same text, so compute it once in a reactive
  # expression and read it with greeting(). The req() here stops both
  # greetings until a name has been entered, since both read greeting().
  greeting <- reactive({
    req(input$name)
    paste0(
      "Hello ",
      input$name,
      ", it is great to know you are ", input$age, " years old!"
    )
  })
  output$greeting <- renderText(greeting())

  # span() escapes its text, so a name such as "<b>Bob</b>" is displayed as
  # typed. Pasting the name into an HTML() string instead would let the user
  # insert their own HTML, or even JavaScript, into the page.
  output$coloured_greeting <- renderUI(
    span(style = paste0("color: ", input$colour, ";"), greeting())
  )
}

shinyApp(ui, server)

library(shiny)
library(tidyverse)

# Read the data once, when the app starts. Shiny runs the app from its own
# folder, so the path is relative to solutions/.
penguins <- read_csv("../src/data/penguins.csv")

# The names are displayed in the dropdowns; the values are the column names
# that input$... will hold.
numeric_choices <- c(
  "Bill length (mm)" = "bill_length_mm",
  "Bill depth (mm)" = "bill_depth_mm",
  "Flipper length (mm)" = "flipper_length_mm",
  "Body mass (g)" = "body_mass_g"
)

categorical_choices <- c(
  "Species" = "species",
  "Island" = "island",
  "Sex" = "sex"
)

# Use the label the user has typed, or `default` if the box is empty.
# (labs(x = "") would remove the axis label altogether.)
label_or_default <- function(label, default) {
  if (label == "") default else label
}

ui <- fluidPage(
  titlePanel("Penguin visualiser"),
  sidebarLayout(
    sidebarPanel(
      selectInput("x_var", "x-axis variable:", choices = numeric_choices),
      selectInput(
        "y_var",
        "y-axis variable:",
        choices = numeric_choices,
        selected = "body_mass_g"
      ),
      selectInput("colour_var", "Colour points by:", choices = categorical_choices),
      textInput("x_label", "x-axis label:"),
      textInput("y_label", "y-axis label:"),
      textInput("title", "Plot title:"),
      selectInput(
        "species",
        "Show species:",
        choices = c("(All species)" = "", "Adelie", "Chinstrap", "Gentoo")
      )
    ),
    mainPanel(
      plotOutput("plot")
    )
  )
)

server <- function(input, output, session) {
  # The filtered data is a reactive expression, so it is only recomputed when
  # the chosen species changes, not when (say) a label is edited.
  selected <- reactive({
    if (input$species == "") {
      penguins
    } else {
      penguins |>
        filter(species == input$species)
    }
  })

  output$plot <- renderPlot({
    # input$x_var etc. are strings, so .data[[...]] is needed to refer to the
    # columns they name.
    ggplot(
      selected(),
      aes(
        x = .data[[input$x_var]],
        y = .data[[input$y_var]],
        colour = .data[[input$colour_var]]
      )
    ) +
      geom_point() +
      # Fix the axes to cover the whole dataset, so that plots of different
      # subsets can be compared, rather than each being rescaled to fit its
      # own points.
      coord_cartesian(
        xlim = range(penguins[[input$x_var]], na.rm = TRUE),
        ylim = range(penguins[[input$y_var]], na.rm = TRUE)
      ) +
      labs(
        # By default, label each axis with the name shown in its dropdown.
        x = label_or_default(input$x_label, names(numeric_choices)[numeric_choices == input$x_var]),
        y = label_or_default(input$y_label, names(numeric_choices)[numeric_choices == input$y_var]),
        colour = names(categorical_choices)[categorical_choices == input$colour_var],
        title = label_or_default(input$title, NULL)
      )
  })
}

shinyApp(ui, server)

library(tidyverse)

penguins <- read_csv("../src/data/penguins.csv")

ui <- fluidPage(
  titlePanel("Penguin visualiser"),
  sidebarLayout(
    sidebarPanel(
      selectInput("x_var", "x-axis variable:", choices = numeric_choices),
      selectInput(
        "y_var",
        "y-axis variable:",
        choices = numeric_choices,
        selected = "body_mass_g"

      selectInput("colour_var", "Colour points by:", choices = categorical_choices),
      textInput("x_label", "x-axis label:"),
      textInput("y_label", "y-axis label:"),
      textInput("title", "Plot title:"),
      selectInput("filter_var", "Filter by:", choices = categorical_choices),
      # The dropdown of values to keep is built in the server, since its
      # choices depend on input$filter_var.
      uiOutput("filter_value_ui")
    ),
    mainPanel(
      plotOutput("plot")
    )
  )

server <- function(input, output, session) {
  output$filter_value_ui <- renderUI({
    values <- sort(unique(na.omit(penguins[[input$filter_var]])))
    selectInput("filter_value", "Show only:", choices = c("(All)" = "", values))
  })

  # The filtered data is a reactive expression, so it is only recomputed when
  # the filter changes, not when (say) a label is edited.
  selected <- reactive({
    # input$filter_value is NULL until the dropdown first appears, and just
    # after a new filter variable is chosen it still holds a value of the old
    # variable (e.g. "Dream" while filtering by sex). Filtering on it then
    # would briefly leave no rows, so only filter on a value that belongs to
    # the current variable; "(All)" is "", which never does.
    if (isTRUE(input$filter_value %in% penguins[[input$filter_var]])) {
      penguins |>
        filter(.data[[input$filter_var]] == input$filter_value)
    } else {
      penguins
    }
  })

  output$plot <- renderPlot({
    # input$x_var etc. are strings, so .data[[...]] is needed to refer to the
    # columns they name.
    ggplot(
      selected(),
      aes(
        x = .data[[input$x_var]],
        y = .data[[input$y_var]],
        colour = .data[[input$colour_var]]
      )
    ) +
      geom_point() +
      # Fix the axes to cover the whole dataset, so that plots of different
      # subsets can be compared, rather than each being rescaled to fit its
      # own points.
      coord_cartesian(
        xlim = range(penguins[[input$x_var]], na.rm = TRUE),
        ylim = range(penguins[[input$y_var]], na.rm = TRUE)
      ) +
      labs(
        # By default, label each axis with the name shown in its dropdown.
        x = label_or_default(input$x_label, names(numeric_choices)[numeric_choices == input$x_var]),
        y = label_or_default(input$y_label, names(numeric_choices)[numeric_choices == input$y_var]),
        colour = names(categorical_choices)[categorical_choices == input$colour_var],
        title = label_or_default(input$title, NULL)
      )
  })
}

library(shiny)
library(tidyverse)

ui <- fluidPage(
  fluidRow(
    column(
      2,
      numericInput("mu1", label = withMathJax("$$\\mu_1$$"), value = -1, width = "80px")
    ),
    column(
      2,
      numericInput("mu2", label = withMathJax("$$\\mu_2$$"), value = 1, width = "80px")
    ),
    column(
      3,
      actionButton("resample", "Draw a new sample")
    )
  ),
  plotOutput("plot"),
  textOutput("means"),
  tableOutput("table")
)

server <- function(input, output, session) {
  # A sample from the standard normal distribution. Only the button changes
  # it, so it is a source that the server sets: a reactiveVal.
  noise <- reactiveVal(tibble(z1 = rnorm(10), z2 = rnorm(10)))

  # Pressing the button should make something happen (draw new noise), so
  # respond to it with an observer.
  observeEvent(input$resample, {
    noise(tibble(z1 = rnorm(10), z2 = rnorm(10)))
  })

  # Adding mu to a standard normal sample gives a sample with mean mu and
  # standard deviation 1. Changing mu1 or mu2 recomputes this, but reuses the
  # same noise, so the points move rather than being redrawn.
  samples <- reactive(tibble(
    x1 = input$mu1 + noise()$z1,
    x2 = input$mu2 + noise()$z2
  ))

  output$plot <- renderPlot(
    {
      ggplot(samples(), aes(x = x1, y = x2)) +
        geom_point() +
        coord_cartesian(xlim = c(-5, 5), ylim = c(-5, 5))
    },
    res = 96
  )

  output$means <- renderText({
    paste(
      "Sample means:",
      round(mean(samples()$x1), 2), "and",
      round(mean(samples()$x2), 2)
    )
  })

  output$table <- renderTable(samples())
}

shinyApp(ui, server)

library(shiny)
library(tidyverse)

ui <- fluidPage(
  fluidRow(
    column(
      2,
      numericInput("mu1", label = withMathJax("$$\\mu_1$$"), value = -1, width = "80px")
    ),
    column(
      2,
      numericInput("mu2", label = withMathJax("$$\\mu_2$$"), value = 1, width = "80px")
    )
  ),
  plotOutput("plot"),
  textOutput("means"),
  tableOutput("table")
)

# Without the reactive expression, each output draws its own sample, so the
# plot, the means and the table all describe different sets of points.
server <- function(input, output, session) {
  output$plot <- renderPlot(
    {
      samples <- tibble(
        x1 = rnorm(10, input$mu1, 1),
        x2 = rnorm(10, input$mu2, 1)
      )
      ggplot(samples, aes(x = x1, y = x2)) +
        geom_point() +
        coord_cartesian(xlim = c(-5, 5), ylim = c(-5, 5))
    },
    res = 96
  )

  output$means <- renderText({
    samples <- tibble(
      x1 = rnorm(10, input$mu1, 1),
      x2 = rnorm(10, input$mu2, 1)
    )
    paste(
      "Sample means:",
      round(mean(samples$x1), 2), "and",
      round(mean(samples$x2), 2)
    )
  })

  output$table <- renderTable({
    tibble(
      x1 = rnorm(10, input$mu1, 1),
      x2 = rnorm(10, input$mu2, 1)
    )
  })
}

shinyApp(ui, server)
