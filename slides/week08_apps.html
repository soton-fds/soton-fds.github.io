Chapter 8: Shiny Demo Apps

These are the four apps from the Chapter 8 lecture. To run one, copy its code into a new .R file in RStudio and click Run App, or use the download link under each app.

The penguin explorer reads data/penguins.csv, relative to the app’s own folder: download the data and save it in a data folder next to the app.

1. Hello Shiny

A slider and a line of text: the smallest useful app. (Download app1_hello.R)

library(shiny)

ui <- fluidPage(
  titlePanel("Hello Shiny!"),
  sidebarLayout(
    sidebarPanel(
      sliderInput("num", "Choose a number:", 1, 100, 50)
    ),
    mainPanel(
      textOutput("result")
    )
  )
)

server <- function(input, output, session) {
  output$result <- renderText({
    paste("You selected:", input$num)
  })
}

shinyApp(ui, server)

2. A conductor

Two outputs that read the same random sample, drawn once in a reactive(). (Download app2_samples.R)

library(shiny)
library(tidyverse)

ui <- fluidPage(
  fluidRow(
    column(
      2,
      numericInput("mu1", label = withMathJax("$$\\mu_1$$"), value = -1, width = "80px")
    ),
    column(
      2,
      numericInput("mu2", label = withMathJax("$$\\mu_2$$"), value = 1, width = "80px")
    ),
    column(
      3,
      textInput("name", label = "Enter your name:", value = "Alan Turing")
    )
  ),
  plotOutput("plot"),
  textOutput("means"),
  textOutput("greeting")
)

server <- function(input, output, session) {
  samples <- reactive(tibble(
    x1 = rnorm(500, input$mu1, 1),
    x2 = rnorm(500, input$mu2, 1)
  ))

  output$plot <- renderPlot(
    {
      ggplot(samples(), aes(x = x1, y = x2)) +
        geom_point() +
        coord_cartesian(xlim = c(-5, 5), ylim = c(-5, 5))
    },
    res = 96
  )

  output$means <- renderText({
    paste(
      "Sample means:",
      round(mean(samples()$x1), 2), "and",
      round(mean(samples()$x2), 2)
    )
  })

  # Does not read samples(), so typing a name never redraws the plot
  output$greeting <- renderText(paste("Hello", input$name))
}

shinyApp(ui, server)

3. A counter

Two buttons that change a reactiveVal() through observeEvent(). (Download app3_counter.R)

library(shiny)

ui <- fluidPage(
  actionButton("inc", "Increase counter"),
  actionButton("reset", "Reset counter"),
  textOutput("value")
)

server <- function(input, output, session) {
  # Create a reactiveVal starting at 0.
  # Retrieve its value by calling it, counter(); set it by calling it with an
  # argument, counter(new_value).
  counter <- reactiveVal(0)

  # Increase counter when 'inc' button is clicked
  observeEvent(input$inc, {
    counter(counter() + 1)   # update value
  })

  # Reset counter when 'reset' button is clicked
  observeEvent(input$reset, {
    counter(0)
  })

  # Display the current value
  output$value <- renderText({
    paste("Counter is:", counter())   # read value
  })
}

shinyApp(ui, server)

4. A penguin explorer

A dropdown, a range slider and a tick-box, a plot and a table, with req() to wait for the user. (Download app4_explorer.R)

library(shiny)
library(tidyverse)

# Read the data once, when the app starts (not inside the server).
penguins <- read_csv("data/penguins.csv")

ui <- fluidPage(
  titlePanel("Penguin explorer"),
  sidebarLayout(
    sidebarPanel(
      selectInput(
        "species",
        "Choose a species:",
        choices = c("(Choose a species)" = "", "Adelie", "Gentoo", "Chinstrap")
      ),
      sliderInput(
        "mass",
        "Body mass (g):",
        min = 2700,
        max = 6300,
        value = c(2700, 6300),
        sep = ""
      ),
      checkboxInput("trend", "Show a trend line", value = TRUE)
    ),
    mainPanel(
      plotOutput("plot"),
      tableOutput("table")
    )
  )
)

server <- function(input, output, session) {
  selected <- reactive({
    penguins |>
      filter(
        species == input$species,
        body_mass_g >= input$mass[1],
        body_mass_g <= input$mass[2]
      )
  })

  output$plot <- renderPlot({
    req(input$species)   # do nothing until a species is chosen
    plot <- ggplot(selected(), aes(x = flipper_length_mm, y = body_mass_g)) +
      geom_point() +
      coord_cartesian(xlim = c(170, 235), ylim = c(2700, 6300)) +
      labs(
        title = paste("Body mass against flipper length:", input$species, "penguins"),
        x = "Flipper length (mm)",
        y = "Body mass (g)"
      )
    if (input$trend) {
      plot <- plot + geom_smooth(se = FALSE)
    }
    plot
  })

  output$table <- renderTable({
    req(input$species)
    selected() |>
      summarise(
        n = n(),
        mean_mass = mean(body_mass_g, na.rm = TRUE)
      )
  })
}

shinyApp(ui, server)
